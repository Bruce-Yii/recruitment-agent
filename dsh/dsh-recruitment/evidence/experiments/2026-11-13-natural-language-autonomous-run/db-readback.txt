-- 数据库读回（ra_s6s7），PII 打码

select left(a.id::text,8) as app, j.code as job, a.current_stage_code as stage, a.lifecycle_status,
       a.record_version as rv,
       left(rv.source_asset_sha256,12) as sha12, left(rv.content_sha256,12) as csha12,
       (select count(*) from recruitment.decision d where d.application_id=a.id) as decisions
from recruitment.application a
join recruitment.job j on j.id=a.job_id
left join recruitment.resume_version rv on rv.id=a.resume_version_id
where a.id='bc2f582b-6b38-48fa-900e-8ffa568f503c';
